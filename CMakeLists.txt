cmake_minimum_required(VERSION 3.10)
project(PointCloudRegistration CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(PCL 1.8 REQUIRED COMPONENTS common io kdtree registration)
add_executable(icp_registration cpp/icp_registration.cpp)
target_include_directories(icp_registration PRIVATE ${PCL_INCLUDE_DIRS})
target_link_directories(icp_registration PRIVATE ${PCL_LIBRARY_DIRS})
target_compile_definitions(icp_registration PRIVATE ${PCL_DEFINITIONS})
target_link_libraries(icp_registration ${PCL_LIBRARIES})
